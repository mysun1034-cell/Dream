## 2. numpy 기초

import numpy as np

arr = np.array([80, 90, 70, 85])
arr.dtype # int64
arr.shape # (4,)

arr2 = np.array([[1, 2, 3],
[4, 5, 6]])
arr2.shape # (2, 3) — 2행 3열
arr2[0, 1] # 0행 1열 → 2
arr2[:, 0] # 모든 행의 0열 → [1, 4]

import pandas as pd

students = pd.DataFrame({

## 4. pandas Part 2 — 선택·필터링·정제·집계

### loc vs iloc

# loc — 라벨(이름) 기반, 슬라이싱 끝 포함
titanic.loc[0]
titanic.loc[0:3, 'survived':'age']

# iloc — 위치(숫자) 기반, 슬라이싱 끝 미포함
titanic.iloc[0]
titanic.iloc[0:3, 0:4]

⚠️ loc는 끝 포함, iloc는 끝 미포함. 자주 헷갈리는 차이!

### 조건 필터링

titanic[titanic['age'] > 35]

# 복합 조건: and/or ❌ → & | ✅ (각 조건에 괄호 필수)
titanic[(titanic['survived'] == 1) & (titanic['sex'] == 'female')]

### 결측치 처리

titanic.isnull().sum() # 컬럼별 결측치 수

titanic_clean = titanic.copy()
titanic_clean['age'] = titanic_clean['age'].fillna(titanic_clean['age'].median())

### 집계

titanic['survived'].value_counts()
titanic['sex'].value_counts(normalize=True) # 비율

titanic.groupby('sex')['survived'].mean()
titanic.groupby('pclass').agg(
생존율=('survived', 'mean'),
인원=('survived', 'count')
)
