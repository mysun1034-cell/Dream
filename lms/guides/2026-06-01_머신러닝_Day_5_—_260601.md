fruits = np.load('fruits_300.npy')

### sklearn KMeans

from sklearn.cluster import KMeans

fruits_2d = fruits.reshape(-1, 100*100) # (300, 10000)

km = KMeans(n_clusters=3, random_state=42)
km.fit(fruits_2d) # y(정답) 없이 fit!

print(km.labels_)
print(np.unique(km.labels_, return_counts=True))
# ([0,1,2], [112, 98, 90])

draw_fruits(fruits[km.labels_ == 0]) # 클러스터 0 시각화
draw_fruits(km.cluster_centers_.reshape(-1, 100, 100), ratio=3) # 중심 이미지

### 주요 속성

|
| 속성 | 설명

| km.labels_ | 각 샘플의 클러스터 번호

| km.cluster_centers_ | 클러스터 중심 (K × features)

| km.inertia_ | 전체 거리 합 (낮을수록 타이트)

| km.n_iter_ | 수렴까지 반복 횟수 (= 4)

| km.transform(X) | 각 centroid까지 거리 행렬

| km.predict(X) | 가장 가까운 클러스터 번호

### 엘보우 기법으로 K 선택

inertia = []
for k in range(2, 7):
km = KMeans(n_clusters=k, random_state=42)
km.fit(fruits_2d)
inertia.append(km.inertia_)

plt.plot(range(2, 7), inertia)
plt.xlabel('k'); plt.ylabel('inertia')
plt.show()

💡 꺾임이 불명확하면 도메인 목적으로 K 결정. 티셔츠 사이즈라면 S/M/L → K=3.

## Step 4. 이상 탐지 (Anomaly Detection)

"Anomaly detection algorithms look at an unlabeled dataset of normal events and learns to detect unusual events." — Andrew Ng

### 핵심 아이디어

K-Means는 거리를 계산한다. 모든 centroid와 멀리 떨어진 점 = 이상치 후보.

dist = km.transform(X) # 각 centroid까지 거리
min_dist = dist.min(axis=1) # 가장 가까운 centroid까지 거리

threshold = np.percentile(min_dist, 95) # 상위 5% = 이상치
is_anomaly = min_dist > threshold
print(f"이상치 후보: {is_anomaly.sum()}개")

### 임계값 선택 trade-off

|
| 퍼센타일 | 이상치 수 | 적합한 상황

| 90% | 많음 | 놓치는 게 더 위험 (사기 탐지)

| 95% | 중간 | 일반적인 기준점

| 99% | 적음 | 과탐지 비용이 클 때 (공장 불량)

### 현실 응용

|
| 도메인 | 정상 패턴 | 이상치

| 금융 | 평소 소비 패턴 | 해외 새벽 고가 결제

| 제조 | 정상 센서 값 | 온도/진동 급변

| IT 보안 | 정상 트래픽 | 갑작스러운 대량 요청

## Step 5. PCA 주성분 분석

왜 필요한가? 10,000개 픽셀 중 상당수는 배경(≈0)이거나 중복 정보. PCA는 핵심 정보만 추출.

비유: 3D 물체를 2D 벽에 투영하면 그림자가 생긴다. PCA는 원본 분산을 가장 잘 설명하는 방향으로 투영.

### PCA 적용

from sklearn.decomposition import PCA

pca = PCA(n_components=50)
pca.fit(fruits_2d)

print(pca.components_.shape) # (50, 10000) — 50개 주성분
fruits_pca = pca.transform(fruits_2d)
print(fruits_pca.shape) # (300, 50) — 압축 완료

### 설명된 분산

print(np.sum(pca.explained_variance_ratio_))
# 0.9215 — 50개로 정보의 92% 보존

plt.plot(pca.explained_variance_ratio_)
plt.show()

# n_components를 비율로도 지정 가능
pca2 = PCA(n_components=0.5)
pca2.fit(fruits_2d)
print(pca2.n_components_) # 2

### PCA + LR 학습 시간 비교

|
| 특성 수 | LR 점수 | 학습 시간

| 원본 10,000개 | 0.9967 | ~1.46s

| PCA 50개 | 0.9967 | ~0.027s ✓

💡 성능 동일, 속도 54배 향상. 10,000개 중 50개만으로 충분합니다.

### PCA 2D + K-Means 시각화

pca = PCA(n_components=0.5)
fruits_pca = pca.fit_transform(fruits_2d)
print(fruits_pca.shape) # (300, 2)

km = KMeans(n_clusters=3, random_state=42)
km.fit(fruits_pca)
# ([0,1,2], [110, 99, 91])

for label in range(3):
data = fruits_pca[km.labels_ == label]
plt.scatter(data[:, 0], data[:, 1])
plt.legend(['cluster 0', 'cluster 1', 'cluster 2'])
plt.show()

### PCA 속성 정리

|
| 속성/메서드 | 설명

| pca.components_ | 주성분 벡터 (n_components × n_features)

| pca.explained_variance_ratio_ | 각 주성분의 분산 기여율

| pca.transform(X) | 고차원 → 저차원 변환

| pca.inverse_transform(X_pca) | 저차원 → 원본 차원 복원 (손실 있음)

| n_components=0.5 | 분산 비율로 주성분 수 자동 결정

|
| 용어 | 뜻 | 비유

| 비지도 학습 | 레이블 없이 데이터 구조를 찾는 학습 | 정답지 없이 비슷한 것끼리 묶기

| 이상 탐지 | 정상 패턴에서 벗어난 데이터 찾기 | 무리에서 혼자 멀리 있는 사람

| PCA | 분산을 최대화하는 방향으로 차원 축소 | 3D 물체를 2D 그림자로 투영

| explained_variance_ratio_ | 각 주성분의 분산 기여율 | 각 축이 얼마나 중요한지 비율

| Adjusted Rand Score | K-Means 결과를 실제 레이블과 비교 (0~1) | 레이블 없이 얼마나 맞췄는지 점수
